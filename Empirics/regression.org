def to_series(df):
    """
    Stack a dataframe to make a series.
    """
    try:
        df.columns.names = [n if n is not None else "_%d" % i for i,n in enumerate(df.columns.names)]
    except AttributeError: # Already a series?
        return pd.Series(df)

    for n in df.columns.names:
        df = df.stack(n)

    return df

def inner(a,b,idxout,colsout,fill_value=None,method='sum'):
    """
    Compute inner product of sorts, summing over indices of products which don't appear in idxout or colsout.
    """

    a = to_series(a)
    b = to_series(b)

    if fill_value is not None:
        a = a.astype(pd.SparseDtype(fill_value=fill_value))
        b = b.astype(pd.SparseDtype(fill_value=fill_value))

    idxint = list(set(a.index.names).intersection(b.index.names))
    aonly = list(set(a.index.names).difference(idxint))
    bonly = list(set(b.index.names).difference(idxint))

    if fill_value is None: # Non-sparse
        a = a.replace(0.,np.nan).dropna()
        b = b.replace(0.,np.nan).dropna()

    c = pd.merge(a.reset_index(aonly),b.reset_index(bonly),on=idxint)
    c = c.reset_index().set_index(idxint + aonly + bonly)

    sumover = list(set(aonly+bonly+idxint).difference(idxout+colsout))
    keep = list(set(aonly+bonly+idxint).difference(sumover))

    if fill_value is not None:
        foo = c.sparse.to_coo().tocsr()

        foo = foo[:,0].multiply(foo[:,1])
        foo = pd.DataFrame.sparse.from_spmatrix(foo,index=c.index)
    else:
        foo = c.iloc[:,0]*c.iloc[:,1]

    if method=='sum':
        p = foo.groupby(keep).sum()
    elif method=='mean':
        p = foo.groupby(keep).mean()
    else:
        raise ValueError("No method %s." % method)

    p = p.unstack(colsout)

    if len(idxout)>1:
        p = p.reorder_levels(idxout)

    p = p.sort_index()

    if len(colsout):
        p = p.sort_index(axis=1)

    return p

** Implementation of $\Mp$

Provide a function =Mp= such that =Mp(X)= returns $X-\E(X|p)$.  In our
case we rely on the law of one price, so that this is a matter of
computing deviations of $X$ from time-market means of $X$.

import pandas as pd
import numpy as np
from warnings import warn

def Mp(X):
    """
    Construct X-E(X|p) = (I-S(S'S)^{-1}S')X.

    Drop any categorical variables where taking means isn't sensible.
    """
    if len(X.shape) > 1:
        X = X.loc[:,X.dtypes != 'category']
    else:
        if X.dtype == 'category': warn('Taking mean of categorical variable.')

    use = list(set(['t','m','j']).intersection(X.index.names))

    if len(use):
        return X - X.groupby(use).transform("mean")
    else:
        return X - X.mean()

** \(\mathcal{M}_d\) operator
Construct =Md= operator.  This can involve strictly categorical variables, in which case we can construct a non-parametric estimator of $\E(X|d=\tau)$ simply by taking means for every value of $\tau$ observed in the sample.    When the method is linear, we instead employ a linear projection.
#+begin_src python
def Md_generator(X,d,method='categorical',expected=False,Mpd=False):
    """
    Md operator, for either categorical or linear expectations.

    If expected is True, return E(X|d) rather than X-E(X|d).
    """
    if method=='categorical':
        try:
            groups = d.columns.tolist()
        except AttributeError: # d is a series?
            if d.name is None: d.name = 'tau'
            groups = [d.name]
        if 'j' in X.index.names: groups += 'j'
        if Mpd: groups += ['t','m']
        groups = list(set(groups)) # Eliminate dupes

        Xg = pd.DataFrame({'X':X}).join(d,how='left').groupby(groups)

        gamma_d = Xg.transform("mean").squeeze()
        gamma_d.name = 'gamma_d'

        if expected: return gamma_d,None

        MdX = X - gamma_d

    elif method=='linear':
        d = pd.DataFrame(d)

        # Add constant column if one doesn't exist in d
        d = d - d.mean()
        d['Constant'] = 1  # Constant just picks up means of X

        if 'j' in X.index.names:
            Xj = X.groupby('j',observed=True)
            if 'j' in d.index.names: dj = d.groupby('j')
            if expected: # Really return gamma_d,gamma
                EdX = {}
                Gamma = {}
                for j,y in Xj:
                    if 'j' in d.index.names:
                        x = dj.get_group(j).droplevel('j')
                    else:
                        x = d
                    EdX[j],Gamma[j] = Md_generator(y.droplevel('j'),x,method='linear',expected=True,Mpd=Mpd)
                EdX = pd.DataFrame(EdX)
                EdX.columns.name = 'j'
                Gamma = pd.DataFrame(Gamma).T
                Gamma.index.name = 'j'

                try:
                    EdX = EdX.stack()
                except AttributeError:
                    pass

                EdX = EdX.reorder_levels(['i','t','m','j']).sort_index()

                return EdX,Gamma
            else:
                MdX = {}
                for j,y in Xj:
                    if 'j' in d.index.names:
                        x = dj.get_group(j).droplevel('j')
                    else:
                        x = d
                    MdX[j] = Md_generator(y.droplevel('j'),x,
                                          method='linear',expected=False,Mpd=Mpd)
                MdX = pd.DataFrame(MdX)
                MdX.columns.name = 'j'

                try:
                    MdX = MdX.stack()
                except AttributeError:
                    pass

                MdX = MdX.reorder_levels(['i','t','m','j']).sort_index()

                return MdX
        else:
            y,x = drop_missing([X,d])

            if Mpd:
                y = y - y.groupby(['t','m']).transform('mean')
                x = x - x.groupby(['t','m']).transform('mean')

            gamma = np.linalg.lstsq(x,y,rcond=None)[0]

            gamma = pd.Series(gamma.squeeze(),index=x.columns)

            if expected:
                # NB: d, not x? Yes, because we can predict even missing values
                # this way.
                gamma_d = (d@gamma).squeeze()
                gamma_d.name = 'gamma_d'

                return gamma_d,gamma
            else:
                gamma_d = (x@gamma).squeeze()
                gamma_d.name = 'gamma_d'

            MdX = pd.Series(y.squeeze() - gamma_d,index=y.index)

    elif method=='mixed': # Some columns of d are categorical, others not
        if expected: raise NotImplementedError
        # Identify categorical columns
        cats = d.select_dtypes(['category']).columns
        xcols = d.columns.difference(cats)

        Z = X.join(d,how='outer')
        ycols = Z.columns.difference(xcols.union(cats))

        # Difference out means for each category
        if len(cats):
            Z = Z - Z.groupby(cats).transform("mean")
        else:
            Z = Z - Z.mean()

        y = Z[ycols]
        x = Z[xcols]

        MdX = Md_generator(y,x,method='linear') # Regress demeaned vars.

    else: raise ValueError("No method %s." % method)

    return MdX

def Mpd_generator(X,tau,method='categorical',Mpd=False):
    """
    Md operator, for either categorical or linear expectations.

    If Mpd is True, operator is Mpd (i.e. condition on d & p jointly).
    """

    if method=='categorical': # assuming conditioning is on groups tau
        groups = ['tau']
        if 'j' in X.index.names: groups += 'j'
        if Mpd: groups += ['t','m']
        Xg = pd.DataFrame({'X':X}).join(tau,how='left').groupby(groups)

        MdX = X - Xg.transform("mean").squeeze()
    elif method=='linear':
        try:
            taucols = tau.columns
            X = pd.DataFrame(X).join(tau,how='outer')
        except AttributeError:  # tau a Series
            taucols = tau

        if 'j' in X.index.names:
            MdX = X.groupby('j').apply(lambda y,x=taucols: Md_generator(y.droplevel('j'),x,method='linear',Mpd=Mpd)).T
            try:
                MdX = MdX.stack()
            except AttributeError:
                pass

            MdX = MdX.reorder_levels(['i','t','m','j']).sort_index()
        else:
            # Difference out kmeans if tau provided
            ycols = X.columns.difference(taucols)
            xcols = taucols
            group = []
            if 'tau' in tau: # kmeans categories provided
                xcols = taucols.drop('tau')
                group += 'tau'
            if Mpd: group += ['t','m']

            if len(group):
                X = X - X.groupby(group).transform("mean")
            else:
                X = X - X.mean()

            y = X[ycols]
            x = X[xcols]
            y,x = drop_missing([y,x])
            x['Constant'] = 1
            b = np.linalg.lstsq(x,y,rcond=None)[0]
            MdX = pd.Series(y.squeeze() - (x@b).squeeze(),index=y.index)
    else: raise ValueError("No method %s." % method)

    return MdX

def estimate_MpMdy(y,d,K=None,Mpd=False):

    if K is not None:
        d  = kmean_controls(K,Mp(y),Mp(d),classifiers=d.loc[:,d.dtypes == 'category'])
        method = 'categorical'
    else:
        method = 'linear'

        # Change categorical vars to numeric
        cats = d.select_dtypes(['category']).columns
        if len(cats):
            d[cats] = d[cats].apply(lambda x: x.cat.codes)

    Md = lambda x: Md_generator(x,d,method=method,Mpd=False)
    if Mpd:
        MpMd = lambda x: Md_generator(x,d,method=method,Mpd=True)
    else:
        MpMd = lambda x: Mp(Md(x))

    MpMdy = MpMd(y)

    assert MpMdy.index.names == ['i','t','m','j']

    if not np.all(np.abs(MpMdy.groupby(['j','t','m']).mean()) < 1e-6):
        warn('MpMdy means greater than 1e-6')

    return MpMdy,Md,MpMd,d

** Estimate \beta from $\MdMpy$

Given $\MpMdy$ estimate \beta.  The default approach to estimation here assumes a form of homoskedasticity.   But this is something we can test (and if necessary, correct).  The basic model involves a mean zero vector $X=\beta w + \epsilon$, from which we construct $\E XX^\T = \beta\E w^2\beta^\T + \E\epsilon\epsilon^\T$.  Under an assumption of homoskedasticity we have $\E\epsilon\epsilon^\T = \sigma^2\mI$.

def estimate_beta(MpMdy,
                  heteroskedastic=False,
                  cov = lambda X : pd.DataFrame.cov(X,ddof=0),
                  return_se=False,bootstrap_tol=None,verbose=False):

    With return_se=False (the default) this is a single cheap SVD.  With
    return_se=True and a bootstrap_tol it runs a resampling bootstrap of at
    least 30 iterations to get standard errors.

    WARNING (nested-bootstrap footgun): do not call with return_se=True
    inside an *outer* whole-pipeline bootstrap -- the inner >=30-iteration
    beta bootstrap then nests, blowing up cost ~30x with no benefit (the
    point estimate is all the outer loop needs).  In bootstrap loops pass
    return_se=False, or call Regression.get_beta with compute_se=False.
    """
    if verbose:
        print("estimate_beta")

    try:
        MpMdY = MpMdy.unstack('j')
    except KeyError:
        MpMdY = MpMdy

    C = cov(MpMdY)

    if np.any(np.isnan(C)):
        raise ValueError(f"Can't compute covariance matrix; too few {C.count().idxmin()}.")

    u,s,vt = svd_missing(C,max_rank=1,heteroskedastic=heteroskedastic)

    if np.sign(u).mean()<0: # Fix sign of u.
        u = -u

    b = pd.DataFrame(u*s,index=MpMdY.columns,columns=['beta'])

    if return_se and bootstrap_tol:
        if bootstrap_tol is None:
            raise ValueError("Not implemented. Specify bootstrap_tol>0.")
            V = (((e-e.mean())**2).mul(v**2,axis=0)).mean() # See p. 150 of Bai (2003)

For each cell $c=(i,t,m)$ the welfare weight is the within-cell
projection of (residualized) log-expenditures onto \beta, computed over
the goods $O_c$ actually observed in that cell:
\[
   \hat w_c = \frac{\beta'X_c}{\beta'\beta}
            = \frac{\sum_{j\in O_c}\beta_j X_{cj}}{\sum_{j\in O_c}\beta_j^2}.
\]
Treating \beta as known (valid for inference on $w$ when $\sqrt{J}/T\to0$,
which holds here since the number of cells far exceeds the number of
goods; see [cite:@bai03]) the sampling variance has the sandwich form
\[
   \widehat{\Var}(\hat w_c)
      = \frac{\sum_{j\in O_c}\beta_j^2 \hat\sigma_{cj}^2}
             {\bigl(\sum_{j\in O_c}\beta_j^2\bigr)^2},
\]
where every sum ranges over the /same/ observed set $O_c$, so the result
is on the same scale as the estimated $\hat w_c$ (i.e. whatever
normalization \beta carries) and is consistent under missing data.  The
earlier expression divided a per-cell good-average by the global
$\beta'\beta$, which silently rescaled the variance by
$\overline{\beta_j^2}=\beta'\beta/J$ and mixed the observed-good count
with the full good count under missing data.

A subtlety the scale fix alone does not address: the residual $e_c$ is
not the raw error but the projection residual $e_c=M_\beta\epsilon_c$, with
$M_\beta=I-\beta\beta'/\beta'\beta$.  Because the $\beta$-direction has
already been fitted out, $\E[e_{cj}^2]=\sigma^2(1-h_{cj})$ with leverage
$h_{cj}=\beta_j^2/\sum_{k\in O_c}\beta_k^2$, so the raw HC0 estimate is
biased /downward/ by a finite-$|O_c|$ factor (about 20--25% at $J=8$;
$\to0$ as $J\to\infty$).  The correction is standard: divide by $(1-h_{cj})$
(HC2), which is exactly unbiased under homoskedasticity and remains
heteroskedasticity-robust.

How the per-good error variance $\sigma_{cj}^2$ is estimated is governed by
the =cov= argument:
- =hc0= (legacy default) :: White single residual, $\hat\sigma_{cj}^2=e_{cj}^2$.  No leverage correction; biased low at finite $J$.  Selected when =cov= is left unset, but with a =DeprecationWarning= --- it is retained as the default only for continuity with existing downstream users, and is slated to change to =hc2=.
- =hc1= :: HC0 scaled by $|O_c|/(|O_c|-1)$.
- =hc2= :: $e_{cj}^2/(1-h_{cj})$ --- exactly unbiased under homoskedasticity, still robust.  Recommended.
- =hc3= :: $e_{cj}^2/(1-h_{cj})^2$; jackknife-style, conservative.
- =homoskedastic= :: a single pooled $\hat\sigma^2=\sum e_{cj}^2/\sum_c(|O_c|-1)$ (proper degrees of freedom), reducing the formula to $\hat\sigma^2/\sum_{j\in O_c}\beta_j^2$.
- =pooled= :: $\hat\sigma_{cj}^2=\Var_i(e)$ within $(t,m,j)$ --- robust to heteroskedasticity across goods and markets, pooled across households so the per-cell estimate is stable (good for inverse-variance weighting downstream).  Not leverage-corrected, so it shares HC0's mild finite-$J$ downward bias.

All options share the same scale and missing-data handling; they differ
only in how $\hat\sigma_{cj}^2$ is estimated and corrected.
#+name: code:wvar

import warnings

def fitted_sigma2(e,cov='homoskedastic'):
    """Per-good error variance sigma_cj^2 estimated from fitted residuals.

    Returns a scalar for cov='homoskedastic' (one pooled sigma^2 with the
    proper d.o.f.) or a wide DataFrame indexed (t,m) with columns j for
    cov='pooled' (Var_i(e) within (t,m,j)).  These are the two w_var
    estimators that do not use the scored cell's own residuals, so they
    are defined for cells outside the fit, including single-good cells;
    Regression.score_w uses them for its w_se diagnostic.
    """
    if cov not in ('homoskedastic','pooled'):
        raise ValueError(f"fitted_sigma2 needs cov 'homoskedastic' or 'pooled', not {cov!r}.")
    E = e.unstack('j') if e.ndim==1 else e
    if cov=='pooled':
        return E.groupby(['t','m'],observed=True).var()
    Jc = E.notna().sum(axis=1)
    dof = (Jc-1).clip(lower=0).sum()                     # pooled, proper d.o.f.
    if dof <= 0:
        warnings.warn("w_var(cov='homoskedastic'): no residual degrees of "
                      "freedom (every cell saturated); returning NaN.")
        return np.nan
    return (E**2).sum().sum()/dof

def w_var(e,beta,cov=None,min_goods=2):
    """Sampling variance of the estimated welfare weights w_hat_c.

    For each cell c=(i,t,m), w_hat_c = (beta'X_c)/(beta'beta) is the
    within-cell projection of residualized log-expenditures onto beta over
    the goods O_c observed in that cell.  Treating beta as known, the
    sandwich variance is

        Var(w_hat_c) = sum_{j in O_c} beta_j^2 sigma_cj^2
                       / (sum_{j in O_c} beta_j^2)^2,

    every sum over the same observed set so the result is on the same scale
    as w_hat_c and is consistent under missing data.

    Because e is the projection residual (the beta-direction is already
    fitted out), E[e_cj^2] = sigma^2 (1 - h_cj) with leverage
    h_cj = beta_j^2 / sum_{k in O_c} beta_k^2; the HC2/HC3 and homoskedastic
    options correct for this finite-J downward bias, HC0/HC1 do not.

    Parameters
    ----------
    e : Series or DataFrame
        Regression residuals (e.g. ``self.e3``); a Series indexed
        ``(i,t,m,j)`` or a wide DataFrame ``(i,t,m) x j``.
    beta : Series
        Frisch elasticities, indexed by good ``j``.
    cov : {'hc0','hc1','hc2','hc3','homoskedastic','pooled'}, optional
        Estimator for the per-good error variance sigma_cj^2.  Leaving it
        unset selects 'hc0' and raises a DeprecationWarning, because that
        default is biased low and is slated to change to 'hc2'; pass cov
        explicitly to silence the warning.

        - 'hc0' (legacy default) : White single residual, sigma_cj^2 = e_cj^2
                            (no leverage correction; biased low at finite J).
        - 'hc1'           : HC0 times |O_c|/(|O_c|-1).
        - 'hc2'           : e_cj^2/(1-h_cj); unbiased under homoskedasticity,
                            still robust.  Recommended.
        - 'hc3'           : e_cj^2/(1-h_cj)^2; conservative.
        - 'homoskedastic' : single pooled sigma^2 with proper d.o.f.
        - 'pooled'        : Var_i(e) within (t,m,j), pooled across
                            households (stable; shares HC0's finite-J bias).

        The 'homoskedastic' and 'pooled' options require residuals from
        which (t,m,j) groups can be formed (a Series, or a wide frame with
        t,m on the row index).

    Returns
    -------
    Series
        Estimated variance of ``w_hat_c`` for each cell ``c=(i,t,m)``, on
        the same scale as the estimated ``w``.
    """
    if cov is None:
        warnings.warn(
            "w_var default cov='hc0' is biased low at finite J (coverage ~0.78 "
            "with complete data, worse under missing goods) and is slated to "
            "change to 'hc2'.  Pass cov explicitly to silence this warning: "
            "'hc2' (recommended) or 'hc0' (legacy behavior).",
            DeprecationWarning, stacklevel=2)
        cov = 'hc0'

    if cov not in ('hc0','hc1','hc2','hc3','homoskedastic','pooled'):
        raise ValueError(f"Unknown cov={cov!r}; expected one of "
                         "'hc0','hc1','hc2','hc3','homoskedastic','pooled'.")

    # Only the homoskedastic estimator (a pooled sigma^2) is defined for
    # single-good cells; the residual-based estimators (hc*, pooled) need
    # >=2 goods per cell and keep the Jc>=2 guard regardless of min_goods.
    if min_goods < 2 and cov != 'homoskedastic':
        raise ValueError(f"min_goods<2 is only valid with cov='homoskedastic', "
                         f"not cov={cov!r} (needs per-cell residual d.o.f.).")

    E = e.unstack('j') if e.ndim==1 else e

    beta = pd.Series(beta).reindex(E.columns)
    if not np.isfinite(beta.astype(float).to_numpy()).all():
        warnings.warn("w_var: beta has NaN/inf for one or more observed goods "
                      "(missing or corrupted elasticities); those goods are dropped.")
    B2 = (E*0).add(beta**2,axis=1)   # beta_j^2 where good observed, NaN where missing
    den = B2.sum(axis=1)             # Sum_{j in O_c} beta_j^2
    Jc = B2.notna().sum(axis=1)      # |O_c|, goods observed in cell
    h = B2.div(den,axis=0)           # leverage h_cj = beta_j^2 / sum_{O_c} beta_k^2

    # Var(w_hat_c) is unestimable when fewer than 2 goods are observed
    # (saturated fit, no residual d.o.f.); such cells return NaN.
    if cov=='homoskedastic':
        sigma2 = fitted_sigma2(E,cov='homoskedastic')
        return (sigma2/den).where(Jc>=min_goods)

    if cov=='pooled':
        # sigma^2_{tmj} pooled across households -> stable per-cell estimate.
        # No leverage correction (it is unstable for small |O_c|), so this
        # shares HC0's mild finite-J downward bias; use 'hc2' if unbiasedness
        # matters more than estimator stability.
        S2 = E.groupby(['t','m']).transform('var')
        valid = E.notna() & np.isfinite(S2)
        num = (B2*S2).where(valid).sum(axis=1)
        d = B2.where(valid).sum(axis=1)
        return (num/d**2).where(Jc>=2)

    WE2 = E.multiply(beta,axis=1)**2                     # beta_j^2 e_cj^2
    if cov=='hc0':
        num = WE2.sum(axis=1)
    elif cov=='hc1':
        num = WE2.sum(axis=1)*(Jc/(Jc-1).where(Jc>=2))
    elif cov in ('hc2','hc3'):
        # An observed good with leverage h_cj==1 (e.g. a 2-good cell where one
        # beta_j==0 -> only one *identifying* good) makes 1-h==0: that cell has
        # no residual d.o.f. for the leverage correction.  NaN the whole cell
        # (not just the good -- skipna would silently drop it) and warn.
        # Only Jc>=2 cells matter here (Jc<2 is already NaN'd below); a
        # leverage-1 good there means a zero-beta good leaving <2 identifying
        # goods.  (A genuine Jc==1 cell trivially has h==1 and is handled by
        # the Jc>=2 guard, so exclude it to avoid a spurious warning.)
        sat = ((h >= 1-1e-12) & E.notna()).any(axis=1) & (Jc>=2)
        if sat.any():
            warnings.warn(f"w_var(cov={cov!r}): {int(sat.sum())} cell(s) have a "
                          "leverage-1 observed good (e.g. beta_j==0 with |O_c|==2); "
                          "no residual d.o.f. for the correction -- returning NaN.")
        omh = (1-h).where(h < 1-1e-12)                   # NaN at saturated goods
        pw = omh if cov=='hc2' else omh**2
        num = (WE2/pw).sum(axis=1).where(~sat)

    return (num/den**2).where(Jc>=2)

** Full covariance of $w$
=w_var= gives only the marginal variances (the diagonal of $V_w$).  For
propagating $w$-uncertainty into a downstream generated-regressor problem
the cross-cell covariance is needed too.  Because the $w$-stage is linear
given \beta --- the joint design is $y_0 = a_{tm} + \beta_j w_{itm} + e$
with an $(t,m)$-intercept block $A$ and a \beta-loaded block $B$ --- the
covariance factors and never needs to be formed as the dense
$N_w\times N_w$ matrix.

Writing $d_c=\sum_{j\in O_c}\beta_j^2$, $s_c=\sum_{j\in O_c}\beta_j$, the
$B$-block normal matrix $B'B=\diag(d_c)$ is diagonal and $A'A=\diag(n_{tm})$
counts observations per market-period.  By the partitioned inverse (FWL),
$\Var(\hat w)=\sigma^2(B'M_AB)^{-1}$ with $M_A=I-A(A'A)^{-1}A'$, and Woodbury
gives the low-rank factoring
\[
   V_w = \sigma^2\bigl[\diag(1/d_c) + U\,V_{22}\,U'\bigr],\quad
   U_{c}=s_c/d_c,\quad
   V_{22,tm}=\frac{1}{\,n_{tm}-\sum_{c\in tm}s_c^2/d_c\,},
\]
where the $V_{22}$ block is /diagonal/ in $(t,m)$ (no cross-market
coupling), so cells are correlated only /within/ a market-period, rank-1
each.  The off-diagonal term is positive: fitting the $(t,m)$ intercept out
inflates $\Var(\hat w)$ above the naive $\sigma^2/d_c$.  This is the
homoskedastic covariance with \beta treated as known; for robust marginals
use =w_var= with a =cov= option.
#+name: code:wcov
#+begin_src python
class WCov:
    """Factored sampling covariance of the welfare-weight estimates w_hat.

    Conditional on beta the w-stage is linear, so V_w is a diagonal plus a
    within-(t,m) rank-1 term and is stored by its factors (the dense
    N_w x N_w matrix is never formed):

        V_w = diag(D) + sigma2 * U V22 U',  cells coupled only within (t,m).

    Attributes
    ----------
    sigma2 : float        pooled error variance
    D : Series (i,t,m)     diagonal part, sigma2 / sum_{O_c} beta_j^2
    U : Series (i,t,m)     loadings s_c/d_c, s_c = sum_{O_c} beta_j
    V22 : Series (t,m)     1 / (n_tm - sum_{c in tm} s_c^2/d_c)
    """
    __slots__ = ['sigma2','D','U','V22']

    def __init__(self,sigma2,D,U,V22):
        self.sigma2,self.D,self.U,self.V22 = sigma2,D,U,V22

    def _v22_by_cell(self):
        return self.V22.reindex(self.U.index.droplevel('i')).values

    def var(self):
        """Marginal variances Var(w_c), the diagonal of V_w."""
        return self.D + self.sigma2*self.U**2*self._v22_by_cell()

    def matvec(self,x):
        """Return V_w @ x without forming V_w (x a Series indexed by cell)."""
        x = x.reindex(self.D.index)
        g = (self.U*x).groupby(['t','m']).sum()*self.V22     # v22_tm * sum_c U_c x_c
        gc = g.reindex(self.U.index.droplevel('i')).values
        return self.D*x + self.sigma2*self.U*gc

    def todense(self,tm):
        """Dense covariance block for one market-period tm=(t,m) (small)."""
        idx = self.U.xs(tm,level=['t','m'],drop_level=False).index
        U = self.U.loc[idx].values; D = self.D.loc[idx].values
        return pd.DataFrame(np.diag(D) + self.sigma2*self.V22[tm]*np.outer(U,U),
                            index=idx,columns=idx)


def w_cov(e,beta,min_goods=2):
    """Factored covariance of w_hat (homoskedastic, beta treated as known).

    Parameters
    ----------
    e : Series or DataFrame
        Regression residuals (e.g. ``self.e3``); Series indexed (i,t,m,j) or
        a wide (i,t,m) x j DataFrame.
    beta : Series
        Frisch elasticities, indexed by good ``j``.
    min_goods : int, default 2
        Cells with fewer than ``min_goods`` observed goods get NaN variance.
        The homoskedastic variance sigma^2/d_c is well-defined for a single
        good (d_c = beta_j^2 > 0), so min_goods=1 includes single-good cells;
        this does NOT change the V22 block (single-good cells contribute
        s_c^2/d_c = 1 against their own n-count, netting to zero), so finite
        Jc>=2 variances are identical whether single-good cells are included
        or excluded.  Default 2 preserves the released behavior.

    Returns
    -------
    WCov
        Factored covariance with ``.var()``, ``.matvec(x)`` and
        ``.todense((t,m))`` helpers.  The diagonal's leading term matches
        ``w_var(e, beta, cov='homoskedastic')``; the extra positive term is
        the cost of estimating the (t,m) intercept.
    """
    E = e.unstack('j') if e.ndim==1 else e
    beta = pd.Series(beta).reindex(E.columns)
    if not np.isfinite(beta.astype(float).to_numpy()).all():
        warnings.warn("w_cov: beta has NaN/inf for one or more observed goods "
                      "(missing or corrupted elasticities); those goods are dropped.")
    obs = E.notna()
    d = obs.multiply(beta**2,axis=1).sum(axis=1)         # d_c = sum_{O_c} beta_j^2
    s = obs.multiply(beta,axis=1).sum(axis=1)            # s_c = sum_{O_c} beta_j
    Jc = obs.sum(axis=1)                                 # |O_c|
    # d>0 excludes cells whose observed goods all have beta==0 (otherwise
    # sigma2/d would be a silent +inf).
    keep = (Jc>=min_goods) & (d>0)

    dof = (Jc-1).clip(lower=0).sum()                     # matches w_var homoskedastic
    if dof <= 0:
        warnings.warn("w_cov: no residual degrees of freedom (every cell saturated); "
                      "sigma2 set to NaN.")
        sigma2 = np.nan
    else:
        sigma2 = (E**2).sum().sum()/dof

    D = (sigma2/d).where(keep)
    U = (s/d).where(keep)

    n_tm = Jc.groupby(['t','m']).sum()                   # total obs per market-period
    corr = (s**2/d).where(d>0).groupby(['t','m']).sum()  # sum_c s_c^2/d_c (saturated cells net to 0)
    den_tm = n_tm - corr                                 # = (A(r) block of B'M_A B), per (t,m)
    # den_tm<=0 means a rank-deficient A(r) block (e.g. an all-saturated
    # market, or perfectly collinear betas): V22 would be +/-inf.  NaN it.
    bad = ~(den_tm > 1e-9*n_tm)
    if bad.any():
        warnings.warn(f"w_cov: {int(bad.sum())} market-period(s) have a rank-deficient "
                      "A(r) block (n_tm - sum_c s_c^2/d_c <= 0); V22 set to NaN there.")
    V22 = (1.0/den_tm).where(~bad)

    return WCov(sigma2,D,U,V22)

def estimate_pi(y,b,w,Ar,gamma_d,verbose=False):

    try:
        y0 = y.stack()
    except AttributeError:
        y0 = y.copy()

    wb = outer(w,b).stack()

    e = y0 - Ar - wb - gamma_d

    e = e.dropna()

    pi_g = e.groupby(['t','m'])

    pi = pi_g.mean()
    pi.name = 'pi'

    pi_se = pi_g.std()/np.sqrt(pi_g.count())

    assert np.all(pi_se>0), "Non-positive estimates of pi_se?!"

    e4 = e - pi
    e4 = e4.reorder_levels(['i','t','m','j']).sort_index()

    return pi, pi_se, e4

def predict_y(pi,Ar,gamma_d,beta,wr):

** Data preparation

Preparation selects the sample used to fit common parameters. Marginal
support uses the same inclusive =min_obs= boundary as the covariance
helper. Goods are removed monotonically, but household eligibility is
reconsidered from the original matched data whenever the goods change.
This permits households to return when the item-fraction denominator falls.
The covariance helper remains a greedy heuristic, not an optimal sample
selector (ledger sections 2, 4, and 5).

The default return remains =(y,d)=. With =return_diagnostics=True= a third
return value reports household and good exclusions and the number of
selection iterations. Inputs are copied, so the caller can retain the
original data for subsequent scoring.

def _prepare_inputs(y,d):
    """Copy and normalize labelled log expenditures and characteristics.

    No sample selection: shared by fitting and frozen-parameter scoring.
    Non-finite logs are unavailable observations; a finite log of zero is
    retained. Duplicate expenditure records are summed in levels.
    """
    if isinstance(y,pd.DataFrame):
        if y.index.names != ['i','t','m'] or y.columns.name != 'j':
            raise ValueError("Wide y must have index (i,t,m) and columns j.")
        # Preserve empty rows and goods in diagnostics on every supported
        # pandas version; legacy stack otherwise drops their observations.
        try:
            y = y.stack('j',future_stack=True)
        except TypeError:  # pandas before future_stack was introduced
            y = y.stack('j',dropna=False)
    else:
        y = y.copy()
    if y.index.names != ['i','t','m','j']:
        raise ValueError("y must have index (i,t,m,j).")
    y = y.astype(float).replace([np.inf,-np.inf],np.nan)
    if not y.index.is_unique:
        grouped = y.groupby(y.index.names,observed=True)
        maxima = grouped.transform('max')
        totals = np.exp(y-maxima).groupby(y.index.names,observed=True).sum(min_count=1)
        y = grouped.max() + np.log(totals)
    y = y.sort_index().rename('y')
    y.index = y.index.remove_unused_levels()

    if 'k' in d.index.names:
        d = d.unstack('k')
    d = pd.DataFrame(d).copy()
    if d.index.names != ['i','t','m']:
        raise ValueError("d must have index (i,t,m) and columns k.")
    if not d.index.is_unique or not d.columns.is_unique:
        raise ValueError("d must have unique household-periods and characteristics.")
    d.columns.name = 'k'
    return y,d.sort_index()


def prepare_data(y,d,min_obs=30,min_prop_items=0.1,alltm=False,
                 return_diagnostics=False):
    """Select fitting observations, reconsidering households as goods fall.

    Returns (y,d), or (y,d,diagnostics) when requested. Diagnostics contains
    household/good DataFrames with retained flags, reasons, and support
    counts, plus the number of selection iterations. It includes all cells
    appearing in either input. No input is changed in place.
    """
    if not isinstance(min_obs,(int,np.integer)) or min_obs < 1:
        raise ValueError("min_obs must be a positive integer.")
    if not 0 <= min_prop_items < 1:
        raise ValueError("min_prop_items must lie in [0,1).")
    y,d = _prepare_inputs(y,d)
    original = y.unstack('j')
    complete = d.notna().all(axis=1)
    numeric = d.select_dtypes(include='number')
    complete &= np.isfinite(numeric).all(axis=1)
    matched = original.loc[original.index.isin(d.index[complete])].dropna(how='all')
    if matched.empty:
        raise ValueError("Join of y & d is empty after removing unavailable observations.")

    goods = matched.columns
    reasons = pd.Series('retained',index=original.columns,dtype=object)
    iterations = 0
    while True:
        iterations += 1
        # Inclusive boundary matches the existing covariance helper (ledger 4).
        Y = matched[goods]
        keep = Y.count() >= min_obs
        reasons.loc[goods[~keep]] = 'min_obs'
        Y = Y.loc[:,keep]
        if alltm:
            keep = (Y.groupby(['t','m'],observed=True).count() > 0).all()
            reasons.loc[Y.columns[~keep]] = 'not_all_markets'
            Y = Y.loc[:,keep]
        if not len(Y.columns):
            raise ValueError("No goods satisfy the fitting support requirements.")

        goods = Y.columns
        # Always reconsider the original matched rows, including prior exclusions.
        Y = Y.loc[Y.count(axis=1) > min_prop_items*len(goods)]
        if Y.empty:
            raise ValueError("No households satisfy min_prop_items.")
        selected = drop_columns_wo_covariance(Y,min_obs=min_obs)
        reasons.loc[goods.difference(selected.columns)] = 'covariance_support'
        if alltm:
            keep = (selected.groupby(['t','m'],observed=True).count() > 0).all()
            reasons.loc[selected.columns[~keep]] = 'not_all_markets'
            selected = selected.loc[:,keep]
        if not len(selected.columns):
            raise ValueError("No goods have adequate paired support in the fitting sample.")
        if selected.columns.equals(goods):
            break
        goods = selected.columns

    fitted_y = selected.stack('j').dropna().rename('y')
    fitted_y.index = fitted_y.index.remove_unused_levels()
    fitted_d = d.reindex(selected.index).copy()
    fitted_d.index = fitted_d.index.remove_unused_levels()
    if not return_diagnostics:
        return fitted_y,fitted_d

    cells = original.index.union(d.index)
    households = pd.DataFrame(index=cells)
    households['n_goods_input'] = original.count(axis=1).reindex(cells,fill_value=0)
    households['n_goods_retained'] = original[goods].count(axis=1).reindex(cells,fill_value=0)
    households['retained'] = cells.isin(selected.index)
    households['reason'] = 'min_prop_items'
    households.loc[households.n_goods_retained == 0,'reason'] = 'no_retained_goods'
    households.loc[households.n_goods_input == 0,'reason'] = 'no_expenditures'
    households.loc[~complete.reindex(cells,fill_value=False),'reason'] = 'missing_characteristics'
    households.loc[households.retained,'reason'] = 'retained'
    goods_report = pd.DataFrame({'n_obs_input':original.count(),
                                'n_obs_matched':matched.count(),
                                'n_obs_retained':selected.count().reindex(original.columns,fill_value=0),
                                'retained':original.columns.isin(goods),
                                'reason':reasons})
    diagnostics = {'households':households,'goods':goods_report,
                   'iterations':iterations}
    return fitted_y,fitted_d,diagnostics

def validate(y,pi,Ar,d,w,beta,gamma,GramSchmidt=False):

        x = x.drop('y',axis=1)

        y,x = drop_missing([y,x])

        b = np.linalg.lstsq(x,y,rcond=None)[0]

    else:

        X = pd.merge(X,gamma_d.reset_index(['i']),left_on=['t','m','j'],right_on=['t','m','j'],how='outer')
    else:
        X = pd.merge(X.reset_index('j'),gamma_d.reset_index(['i']),left_on=['t','m'],right_on=['t','m'],how='outer')

    X = X.rename(columns={('i',''):'i'}) # Deal with bug in reset_index for sparse matrices?

    X = X.reset_index().set_index(['i','t','m','j'])

    w.name='w'

    bw = outer(w,beta).stack()
    bw.name = 'bw'

    if GramSchmidt:
        MpMd = lambda x: Md_generator(x,d,Mp=True)
        bw = Mp(MpMd(bw))
        bw.name = 'bw'

    X = X.join(bw[y.index])

    X['y'] = y
    X = X.dropna()
    X.columns.name = 'l'

    B = X.groupby('j').apply(lambda x: ols(x))

    return B,X

        inner >=30-iteration bootstrap.  When resampling the whole pipeline

        path) so the beta bootstrap does not nest ~30x inside each outer

    def w_se(self,cov=None):
        """Compute standard errors of estimated ws.

        NOTE: leaving ``cov`` unset selects the legacy ``'hc0'`` estimator and
        raises a DeprecationWarning.  HC0 is biased *low* at finite J because
        the w-residuals come from a projection (the beta-direction is fitted
        out): coverage ratios run ~0.78 with complete data and worse under
        missing goods (~0.6 at 30% missing, ~0.4 at 50%).  It is retained as
        the default only for continuity with existing downstream code.  Pass
        ``cov='hc2'`` for the leverage-corrected estimator, which is unbiased
        and remains heteroskedasticity-robust.

        - cov :: Per-good error-variance estimator passed to ``w_var``; one
          of 'hc0' (legacy default), 'hc1', 'hc2' (recommended), 'hc3',
          'homoskedastic', or 'pooled'.
        """
        if cov is None:
            warnings.warn(
                "w_se default cov='hc0' is biased low at finite J and is slated "
                "to change to 'hc2'.  Pass cov explicitly to silence this "
                "warning: 'hc2' (recommended) or 'hc0' (legacy behavior).",
                DeprecationWarning, stacklevel=2)
            cov = 'hc0'

        beta = self.get_beta()

        if self.e3 is None: self.get_w()

        v = w_var(self.e3,beta,cov=cov)

        return np.sqrt(v)

    def score_w(self,y,d,min_goods=1,return_diagnostics=False,cov='homoskedastic'):
        """Estimate welfare for additional households using this fitted model.

        Parameters
        ----------
        y : Series (i,t,m,j) or DataFrame (i,t,m) x j
            Log expenditures. Non-finite entries are unavailable; finite
            zero logs are valid. Duplicate records are summed in levels.
        d : DataFrame (i,t,m) x k or Series (i,t,m,k)
            Numeric characteristics with the same meaning as in the fit.
            Extra columns are ignored; required columns must be present.
        min_goods : int, default 1
            Minimum number of observed fitted goods for a score. A positive
            sum of squared loadings is also required.
        return_diagnostics : bool, default False
            Return (w,diagnostics), with observed/usable-good counts,
            sum_beta2, a sampling standard error w_se, and a status for
            every cell in either input.
        cov : {'homoskedastic','pooled'}, default 'homoskedastic'
            Error-variance estimator behind w_se, taken from the fitting
            residuals (see fitted_sigma2 and w_var).  The hc* estimators
            of w_var are not offered: they need the cell's own residuals,
            which a single-good cell does not have.

        Returns
        -------
        Series (i,t,m), or (Series, DataFrame)
            Scores on the fitted normalization; NaN for unscoreable cells.
            Parameters are fitted lazily from self.y/self.d if needed,
            never from these new observations. Only linear numeric controls
            and fitted market-periods are currently supported. The w_se
            diagnostic treats beta, gamma and the market offsets as known,
            exactly as w_var does for the fitting sample; it carries no
            parameter-estimation or selection uncertainty. With one good
            w_se is sigma/|beta_j|, several times the fitting-sample se for
            typical elasticities, so filter on w_se rather than on n_goods
            alone. Under cov='pooled', w_se is NaN for a scored cell none of
            whose observed goods has a training variance in its (t,m).
        """
        if not isinstance(min_goods,(int,np.integer)) or min_goods < 1:
            raise ValueError("min_goods must be a positive integer.")
        if cov not in ('homoskedastic','pooled'):
            raise ValueError(f"cov must be 'homoskedastic' or 'pooled', not {cov!r}.")
        if self.attrs.get('method','linear') != 'linear' or self.attrs.get('K') is not None:
            raise NotImplementedError("score_w currently supports linear numeric controls with K=None.")
        if self.d is None or any(not pd.api.types.is_numeric_dtype(x) for x in self.d.dtypes):
            raise NotImplementedError("score_w requires numeric fitting characteristics.")
        y,d = _prepare_inputs(y,d)
        self.get_w()
        beta = self.beta
        gamma = self.get_gamma()
        if not np.isfinite(beta.to_numpy()).all() or not np.isfinite(gamma.to_numpy()).all():
            raise ValueError("Fitted beta and gamma must be finite.")

        # Ed centers controls on the fitting sample (ledger 2 and 4).
        features = gamma.columns.drop('Constant')
        missing = features.difference(d.columns)
        if len(missing):
            raise ValueError(f"Missing scoring characteristics: {missing.tolist()}.")
        if any(not pd.api.types.is_numeric_dtype(x) for x in d[features].dtypes):
            raise NotImplementedError("score_w requires numeric scoring characteristics.")
        Y = y.unstack('j')
        cells = Y.index.union(d.index)
        # Float conversion makes nullable numeric missing values ordinary
        # NaNs. With no controls, completeness is true for every scoring cell.
        controls = d[features].reindex(cells).astype(float)
        complete = np.isfinite(controls).all(axis=1)
        design = controls - self.d[features].mean()
        design['Constant'] = 1.
        expected = design[gamma.columns] @ gamma.T

        # Recover the intercept in the existing joint fit, including its
        # Mpi centering. Ar is a post-fit summary, not this intercept (ledger 4).
        train_index = self.y.index
        bw = (self.w.reindex(train_index.droplevel('j')).to_numpy() *
              beta.reindex(train_index.get_level_values('j')).to_numpy())
        training_residual = self.y - self.gamma_d.reindex(train_index) - bw
        offset = training_residual.groupby(['t','m'],observed=True).mean()

        n_observed = Y.count(axis=1).reindex(cells,fill_value=0)
        Y = Y.reindex(index=cells,columns=beta.index)
        market = cells.droplevel('i')
        fitted_offset = pd.Series(offset.reindex(market).to_numpy(),index=cells)
        residual = Y - expected.reindex(cells)
        residual = residual.sub(fitted_offset,axis=0)
        observed = Y.notna()
        n_goods = observed.sum(axis=1)
        sum_beta2 = observed.mul(beta**2,axis=1).sum(axis=1)
        numerator = residual.mul(beta,axis=1).sum(axis=1,min_count=1)
        w = numerator / sum_beta2.where(sum_beta2 > 0)

        status = pd.Series('scored',index=cells,dtype=object)
        status.loc[sum_beta2 <= 0] = 'zero_loading'
        status.loc[n_goods < min_goods] = 'too_few_goods'
        status.loc[n_goods == 0] = 'no_supported_goods'
        status.loc[n_observed == 0] = 'no_expenditures'
        status.loc[~np.isfinite(fitted_offset)] = 'unknown_market'
        status.loc[~complete] = 'missing_characteristics'
        status.loc[(status == 'scored') & ~np.isfinite(w)] = 'nonfinite_score'
        w = w.where(status == 'scored').rename('w')
        if not return_diagnostics:
            return w
        # Sampling variance of the score with beta known (the w_var sandwich):
        # sum_{j in O_c} beta_j^2 sigma_cj^2 / (sum_{j in O_c} beta_j^2)^2,
        # with sigma^2 from the fitting residuals so it is defined off-sample.
        sigma2 = fitted_sigma2(self.e3,cov=cov)
        if cov=='pooled':
            S2 = sigma2.reindex(index=market,columns=beta.index)
            S2 = pd.DataFrame(S2.to_numpy(),index=cells,columns=beta.index)
            valid = observed & np.isfinite(S2)
            B2 = observed.mul(beta**2,axis=1)
            var = (B2*S2).where(valid).sum(axis=1)/B2.where(valid).sum(axis=1)**2
        else:
            var = sigma2/sum_beta2.where(sum_beta2 > 0)
        w_se = np.sqrt(var).where(status == 'scored').rename('w_se')
        diagnostics = pd.DataFrame({'n_observed':n_observed,'n_goods':n_goods,
                                    'sum_beta2':sum_beta2,'w_se':w_se,'status':status})
        return w,diagnostics

    def predicted_expenditures(self,resample=False,clusterby=['t','m','j'],fill_missing=True,verbose=False):
        """Compute predicted /levels/ of expenditures.

        This is different from exp(yhat), since we have to account for the expected value of exp(e), where e = y - yhat.

        One standard (and the default) way to calculate these is by assuming that the distribution of e is normal.  An alternative is to resample residuals.

        Regardless of whether resampling is chosen, means (and variances) are selected at the level of the list clusterby.
        """
        if self.yhat is None:
            self.yhat = self.get_predicted_log_expenditures(fill_missing=True,verbose=verbose)

        e = self.y - self.yhat
        eg = e.dropna().groupby(clusterby)

        if not resample:
            # Use iqr instead of variance for some robustness to outliers
            # Relation for normal dist: iqr/1.3489795 = sigma
            evar = (eg.transform(iqr)/1.3489795)**2
        else:
            if resample < 1: # Assume this is a tolerance
                tol = resample
            last = -1
            evar = e.dropna().groupby(clusterby).transform(np.var)
            evar = evar.sort_index(level=clusterby)
            i = 0
            diff = 1
            while diff>tol:
                last = evar
                esample = eg.sample(frac=1,replace=True)
                drawvar = (esample.groupby(clusterby).transform(iqr)/1.3489795)**2
                evar = i/(i+1)*evar + drawvar.values/(i+1)
                i += 1
                diff = np.abs(evar-last).max()

*** Predicted \lambda
Here we want estimates of \lambda, or the antilog of $-w$.  We have a CLT for $w$ [cite:@bai03] and an expression for the asymptotic variance of $w$, say $\hat\sigma^2$.  Accordingly, we should have $\lambda = e^{-w}\approx \E e^{-\hat w + \hat\sigma^2/2}$.

    def predicted_lambda(self):
        """Expected value of minus the antilog of $w$.

        v = self.w_se()**2
        lhat = np.exp(-self.w + v/2)

        return lhat

*** Average Estimator of =w=
It's possible to estimate =w= without first estimating \beta, etc. or using factor analysis.  We simply compute =Mpdy= (residuals from the SUR regression), then compute the mean across $j$ for every $(i,t,m)$.  The defect in this simple plan is that it relies on all households reporting positive expenditures for the same number of goods.  Still, this may not matter than much in practice.  It should be more efficient to weight by the reciprocal of the standard deviation of errors for each good, using a GLS sort of logic; this is made an option.

    def average_to_get_w(self,weight=False):
        e = self.get_Mpdy()
        if weight:
            sigmaj = e.groupby('j').transform('std')
        else:
            sigmaj = 1
        w = (e/sigmaj).groupby(['i','t','m']).mean()

        return w

def _demand_parameters(self,p=None,d=None):
    """Return tuple of p and dictionary of (alpha,beta,phi) from regression instance.

    Suitable for passing to =cfe.demand= functions.
    """

    beta = self.beta
    n = len(beta)

    if d is None:
        gd = self.get_gamma_d().groupby('j').mean()
    else:
        gd = d@self.gamma.T

    alpha = np.exp(gd)

    if p is None or len(p)==0:
        prices = np.exp((self.pi + self.Ar).groupby('j').mean())
    else:
        prices = p

    assert len(prices), f"What happened to prices? p={prices}."

    phi = 0 # phi not (yet?) an attribute of Regression.

    # Cast to numeric arrays
    try:
        prices = prices.values
    except AttributeError:
        pass

    try:
        alpha = alpha.values
    except AttributeError:
        pass
    try:
        beta = beta.values
    except AttributeError:
        pass

    try:
        phi = phi.values
    except AttributeError:
        pass

    return prices,{'alpha':alpha,'beta':beta,'phi':phi}

def _demands(self,x,p=None,z=None,type="Marshallian"):
    """Quantities demanded at prices p for household with observable
    characteristics z, having a utility function with parameters given
    by (possibly estimated) attributes from a Regression (i.e., the
    vectors of parameters alpha, beta, delta).

    Default type is "Marshallian", in which case argument x is budget.

    Alternative types:
       - "Frischian" :: argument x is Marginal utility of expenditures
       - "Hicksian" :: argument x is level of utility

    Ethan Ligon                                    April 2019
    """

    idx = self.get_beta().index
    p,pparms = _demand_parameters(self,p,z)

    Qs = {'Marshallian':consumerdemands.marshallian.demands,
          'Hicksian':consumerdemands.hicksian.demands,
          'Frischian':consumerdemands.frischian.demands}

    q = pd.Series(Qs[type](x,p,pparms),index=idx,name='quantities')

    return q

    """Expenditures for different goods at prices p for household with observable
    characteristics z, having a utility function with parameters given
    by (possibly estimated) attributes from a Regression (i.e., the
    vectors of parameters alpha, beta, delta).

    Default type is "Marshallian", in which case argument x is budget.

    Alternative types:
       - "Frischian" :: argument x is Marginal utility of expenditures
       - "Hicksian" :: argument x is level of utility

def read_pickle(fn,cache_dir=None):
    """
    Read pickled dictionary and assign keys as attributes to Regression object.
    """
    import fsspec

    try:
        R = pickle.load(fn)  # Is fn a file?
    except TypeError:  # Maybe a filename?
        if cache_dir is not None:
            if 'filecache::' not in fn:  # May already have caching specified
                fn = 'filecache::' + fn
            storage_options = {'filecache':{'cache_dir':cache_dir}}
            with fsspec.open(fn,mode='rb',
                             storage_options=storage_options) as f:
                R = pickle.load(f)
        else:
            with fsspec.open(fn,mode='rb') as f:
                R = pickle.load(f)

    if type(R) is not dict:
        R = R.__dict__
        # Fix ill-considered attribute name
        try:
            R['mults_se'] = R['se_mult']
            del R['se_mult']
        except KeyError:
            pass

    return Regression(**R)

** Preparation and scoring

=Regression.preparation= records selection diagnostics. To recover scores
for households excluded from fitting, retain the original inputs and call
=r.score_w(y,d,return_diagnostics=True)= after fitting =r=. The method
returns scores and support diagnostics for cells appearing in either
input, including explicit reasons for unavailable scores. A single observed
fitted good can suffice, conditional on a nonzero loading and fitted
controls and market effects. The training item-fraction cutoff is not
applied to scoring.

This first interface supports linear numeric controls and existing
market-periods. Its centering and market intercept come entirely from the
fit. In particular, the post-fit =Ar= summary must not be substituted for
the market intercept of =estimate_w=. The tests check agreement with the
original estimator, scoring independent of batch composition, persistence,
and sample restoration. They do not assert coverage for an uncertainty
estimator with estimated nuisance parameters.

The diagnostics also carry =w_se=, the =w_var= sandwich with $\beta$ known
and $\hat\sigma^2$ taken from the fitting residuals (=cov='homoskedastic'=
or =cov='pooled'=; see =fitted_sigma2=). Because $\hat\sigma^2$ does not
come from the scored cell, =w_se= is defined for every scored cell,
including single-good ones, and on the fitting sample it coincides with
=w_var= under the same =cov=. It is the quantity to filter on: the training
cutoff =min_prop_items= picks a point on the precision curve, and a scored
cell with one good has $\hat\sigma/|\beta_j|$, which for typical
elasticities is several times the fitting-sample standard error.

** Tests for =w_var=
These guard the welfare-weight sampling variance: its scale (sums over the
observed good set $O_c$, not the global $\beta'\beta$), its missing-data
handling, and the finite-$J$ coverage behavior of the =cov= options.

import numpy as np
import pandas as pd
import pytest
from cfe.regression import w_var, w_cov, outer


def _cells(rows):
    return pd.DataFrame(
        rows,
        index=pd.MultiIndex.from_tuples([(i,0,0) for i in range(len(rows))],
                                        names=['i','t','m']),
        columns=pd.Index(range(len(rows[0])), name='j'),
    ).stack()


def _messy_w_model(N=120, J=6, miss=0.3, seed=5):
    # Missing goods + noise -> realistic Jc heterogeneity in the residuals.
    rng = np.random.default_rng(seed)
    beta = pd.Series(np.linspace(.4,3,J), index=pd.Index(range(J),name='j'))
    cells = pd.MultiIndex.from_arrays(
        [np.arange(N), np.zeros(N,int), np.zeros(N,int)], names=['i','t','m'])
    w = pd.Series(rng.standard_normal(N), index=cells)
    aj = rng.standard_normal(J); aj -= aj.mean()
    Ar = pd.Series(aj, index=pd.MultiIndex.from_tuples(
        [(0,0,j) for j in range(J)], names=['t','m','j']))
    bw = outer(w,beta).stack(); bw.index.names=['i','t','m','j']
    y = (bw + pd.Series(Ar.reindex(bw.index.droplevel('i')).values, index=bw.index)
         + rng.standard_normal(len(bw))*0.4)
    y = y[rng.random(len(y)) >= miss]
    return (y - 0*y), beta            # (residual-like, beta) for w_var/w_cov


def test_hc0_matches_sandwich_scale():
    # Corrected HC0 == OLS-slope sandwich for w_hat_c = (beta'X_c)/(beta'beta).
    beta = pd.Series([1.,2.,3.], index=pd.Index([0,1,2],name='j'))
    e = _cells([[0.1,-0.2,0.05],[0.0,0.1,-0.1]])
    v = w_var(e, beta, cov='hc0')
    E = e.unstack('j').values; b = beta.values; b2 = (b**2).sum()
    expect = (b**2*E**2).sum(1)/b2**2
    assert np.allclose(v.values, expect)
    old = ((E*b)**2).mean(1)/b2          # old (global-betabeta, mean-over-j) formula
    assert not np.allclose(v.values, old)


def test_hc2_leverage_correction():
    beta = pd.Series([1.,2.,3.], index=pd.Index([0,1,2],name='j'))
    e = _cells([[0.1,-0.2,0.05]])
    b = beta.values; b2 = (b**2).sum(); h = b**2/b2
    E = e.unstack('j').values
    expect = (b**2*E**2/(1-h)).sum(1)/b2**2
    assert np.allclose(w_var(e, beta, cov='hc2').values, expect)


def test_missing_data_uses_observed_set():
    beta = pd.Series([1.,2.,3.], index=pd.Index([0,1,2],name='j'))

def test_coverage_hc2_unbiased_hc0_low():
    rng = np.random.default_rng(0)
    J, N, sigma = 8, 40000, 0.7
    beta = np.linspace(.2,5,J); b2 = beta@beta
    w = rng.standard_normal(N)
    X = np.outer(w,beta) + rng.standard_normal((N,J))*sigma
    what = X@beta/b2
    e = X - np.outer(what,beta)
    emp = np.var(what - w)
    cells = pd.MultiIndex.from_arrays(
        [np.arange(N), np.zeros(N,int), np.zeros(N,int)], names=['i','t','m'])
    eS = pd.DataFrame(e, index=cells, columns=pd.Index(range(J),name='j')).stack()
    bS = pd.Series(beta, index=pd.Index(range(J),name='j'))
    assert 0.95 < w_var(eS,bS,cov='hc2').mean()/emp < 1.06
    assert 0.95 < w_var(eS,bS,cov='homoskedastic').mean()/emp < 1.06
    assert w_var(eS,bS,cov='hc0').mean()/emp < 0.85


def _brute_force_w_cov(eS, beta, tmlist):
    # Build the joint design [A | B] and form sigma^2 (B'M_A B)^{-1}.
    E = eS.unstack('j'); cells = E.index; J = E.shape[1]
    b = beta.reindex(E.columns).values
    nobs = len(cells)*J
    A = np.zeros((nobs, len(tmlist))); B = np.zeros((nobs, len(cells)))
    r = 0
    for ci,(i,t,m) in enumerate(cells):
        for jj in range(J):
            A[r, tmlist.index((t,m))] = 1.0
            B[r, ci] = b[jj]
            r += 1
    MA = np.eye(nobs) - A@np.linalg.inv(A.T@A)@A.T
    sigma2 = w_cov(eS, beta).sigma2
    return sigma2*np.linalg.inv(B.T@MA@B), cells


def test_w_cov_matches_partitioned_inverse():
    rng = np.random.default_rng(1)
    J, Ncell = 5, 16
    beta = np.linspace(.5,3,J)
    e = rng.standard_normal((Ncell,J))*0.6; e = e - e.mean(0)
    cells = pd.MultiIndex.from_arrays(
        [np.arange(Ncell), np.zeros(Ncell,int), np.zeros(Ncell,int)], names=['i','t','m'])
    eS = pd.DataFrame(e, index=cells, columns=pd.Index(range(J),name='j')).stack()
    bS = pd.Series(beta, index=pd.Index(range(J),name='j'))
    W = w_cov(eS, bS)
    Vexact, _ = _brute_force_w_cov(eS, bS, [(0,0)])
    assert np.allclose(W.todense((0,0)).values, Vexact)
    assert np.allclose(W.var().values, np.diag(Vexact))
    x = pd.Series(rng.standard_normal(Ncell), index=cells)
    assert np.allclose(W.matvec(x).values, Vexact@x.values)
